## A project in three files

```ts title="docs.config.ts"
import { defineConfig } from "@consify/core";
import { docs } from "@consify/docs";

export default defineConfig({
  site: { name: "My project", github: { repo: "me/my-project" } },
  i18n: { languages: ["en", "ru"] },
  features: [docs()],
});
```

```ts title="vite.config.ts"
import { consify } from "@consify/core/vite";
import { defineConfig } from "vite";
import config from "./docs.config.ts";

export default defineConfig({ plugins: [consify(config)] });
```

```ts title="react-router.config.ts"
import { defineRouterConfig } from "@consify/core/react-router";
import config from "./docs.config.ts";

export default defineRouterConfig(config);
```

Then write pages in `content/en/docs/`, run `bun run dev`, and you have a site. Read [How it works](/docs/v0/concepts) to see what happens in between, or [how consify was made](/blog/how-consify-was-made) to see why it looks like this.
