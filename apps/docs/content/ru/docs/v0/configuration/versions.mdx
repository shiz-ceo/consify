```ts title="docs.config.ts"
import { defineConfig } from "@consify/core";
import { docs } from "@consify/docs";

export default defineConfig({
  site: { name: "My docs" },
  features: [
    docs({
      versions: {
        list: [
          { id: "v2", label: "v2 (latest)", status: "latest" },
          { id: "v1", label: "v1", status: "deprecated" },
        ],
        default: "v2",
      },
    }),
  ],
});
```

`versions` — опция `docs()` (см. [Документация](/docs/v0/configuration/docs)). Без неё документация лежит прямо в `content/en/docs`, а переключателя нет.

## Как выглядят папки

Каждая версия — это папка с `"root": "version"` в её `meta.json`:

```
content/en/docs/
├─ meta.json            { "pages": ["v2", "v1"] }
├─ v2/
│  ├─ meta.json         { "title": "v2 (latest)", "root": "version", ... }
│  └─ index.mdx
└─ v1/
   ├─ meta.json
   └─ index.mdx
```

`/en/docs` перенаправляет на версию по умолчанию. Переключатель оставляет читателя на той же странице, если она есть в другой версии.
