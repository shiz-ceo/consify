<Step>

### Describe the job

The type argument is the contract between producer and worker.

```ts twoslash title="queue.ts"
import { createQueue } from "lattice";

export interface ResizeImage {
  url: string;
  width: number;
}

export const images = createQueue<ResizeImage>("images");
```

</Step>
<Step>

### Add a job

`add` resolves when the job is stored. Passing data of the wrong shape is a compile error, not a runtime surprise:

```ts twoslash
import { createQueue } from "lattice";
interface ResizeImage { url: string; width: number }
const images = createQueue<ResizeImage>("images");
// ---cut---
await images.add({ url: "https://example.com/a.png", width: 640 });

// @errors: 2353
await images.add({ url: "https://example.com/a.png", height: 480 });
```

</Step>
<Step>

### Process it

```ts twoslash title="worker.ts"
import { createQueue, createWorker } from "lattice";
interface ResizeImage { url: string; width: number }
const images = createQueue<ResizeImage>("images");
// ---cut---
const worker = createWorker(
  images,
  async (job) => {
    console.log(`Resizing ${job.data.url} to ${job.data.width}px`);
  },
  { concurrency: 4 },
);

process.on("SIGTERM", () => worker.close());
```

</Step>
</Steps>
